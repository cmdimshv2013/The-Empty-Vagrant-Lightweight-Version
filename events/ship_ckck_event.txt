namespace = ckck_event

country_event = {
	id = ckck_event.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		fromfrom = {
			check_modifier_value = { modifier = sg_instant_kill_shot_add value > 0 }
			NAND = { is_variable_set = sg_instant_kill_shot_finished check_variable = { which = sg_instant_kill_shot_finished value >= modifier:sg_instant_kill_shot_add } }
		}
	}
	immediate = {
		fromfrom = {
				if = { limit = { NOT = { is_variable_set = sg_instant_kill_shot_finished } } set_variable = { which = sg_instant_kill_shot_finished value = 1 } }
				else = { change_variable = { which = sg_instant_kill_shot_finished value = 1 } }
		}
	}
}
